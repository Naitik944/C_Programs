#include <stdio.h>
#include "addition.h"
#include "subtract.h"
#include "multiply.h"
#include "divide.h"

int main(void) {
    int num1, num2;
    char operation;

    printf("Enter your first number: ");
    scanf("%d", &num1);

    printf("Enter your second number: ");
    scanf("%d", &num2);

    printf("Choose your operation (+, -, *, /): ");
    scanf(" %c", &operation);

    switch (operation) {
        case '+':
            printf("Sum of the given numbers is : %d\n", add(num1, num2));
            break;

        case '-':
            printf("Difference of the given numbers is : %d\n", subtract(num1, num2));
            break;

        case '*':
            printf("Product of the given numbers is : %d\n", multiply(num1, num2));
            break;

        case '/':
            if (num2 == 0) {
                printf("Division by zero is undefined\n");
                return 1;
            }
            printf("Quotient of the given numbers is : %d\n", divide(num1, num2));
            printf("Remainder of the given numbers is : %d\n", calculate_remainder(num1, num2));
            break;

            default:
               printf("Unsupported operation.\n");
            return 1;
    }
    return 0;
}
